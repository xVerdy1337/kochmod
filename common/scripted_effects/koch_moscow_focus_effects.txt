# The event saves the player's intended side separately from each country's branch.
KOCH_B07_commit_side_choice = {
    if = {
        limit = {
            tag = B07
            NOT = { has_country_flag = KOCH_B07_side_choice_locked }
            NOT = { has_country_flag = KOCH_B07_civil_war_started }
        }
        set_country_flag = KOCH_B07_side_choice_locked
        KOCH_B07_initialize_balance = yes
        clr_country_flag = KOCH_B07_side_choice_pending
    }
}

# Reduce tension once; never affect post-war or resolved-crisis setup.
KOCH_B07_delay_democratic_crisis = {
    if = {
        limit = {
            tag = B07
            NOT = { has_country_flag = KOCH_B07_civil_war_started }
            has_country_flag = KOCH_B07_crisis_active
            NOT = { has_country_flag = KOCH_B07_crisis_resolved }
            NOT = { has_country_flag = KOCH_B07_democratic_delay_applied }
        }
        set_country_flag = KOCH_B07_democratic_delay_applied
        subtract_from_variable = { KOCH_B07_tension = 15 }
        KOCH_B07_check_tension = yes
    }
}

# Resetting inherited choices cannot repeat the guarded tension reduction.
KOCH_B07_clear_inherited_side_focuses = {
    if = {
        limit = {
            has_completed_focus = KOCH_B07_democratic_path
            NOT = { has_country_flag = KOCH_B07_democratic_side }
        }
        uncomplete_national_focus = { focus = KOCH_B07_democratic_path uncomplete_children = yes }
    }
    if = {
        limit = {
            has_completed_focus = KOCH_B07_fascist_path
            NOT = { has_country_flag = KOCH_B07_fascist_side }
        }
        uncomplete_national_focus = { focus = KOCH_B07_fascist_path uncomplete_children = yes }
    }
    if = {
        limit = {
            has_completed_focus = KOCH_B07_imperial_path
            NOT = { has_country_flag = KOCH_B07_authoritarian_side }
        }
        uncomplete_national_focus = { focus = KOCH_B07_imperial_path uncomplete_children = yes }
    }
}

KOCH_B07_assign_civil_war_focus_paths = {
    if = {
        limit = {
            tag = B07
            has_country_flag = KOCH_B07_civil_war_started
            NOT = { has_country_flag = KOCH_B07_focus_paths_assigned }
            country_exists = KMF
            country_exists = KMA
        }
        set_country_flag = KOCH_B07_focus_paths_assigned
        set_country_flag = KOCH_B07_democratic_side
        clr_country_flag = KOCH_B07_fascist_side
        clr_country_flag = KOCH_B07_authoritarian_side

        # Copy the common pre-war progress before changing B07's selected path.
        KMF = { load_focus_tree = { tree = KOCH_moscow_shared_tree copy_completed_from = B07 } }
        KMA = { load_focus_tree = { tree = KOCH_moscow_shared_tree copy_completed_from = B07 } }
        load_focus_tree = { tree = KOCH_moscow_shared_tree keep_completed = yes }

        # Tension can reach 100 even if nobody researched the choice focus.
        # Its reward is guarded against opening the event after the split.
        set_country_flag = KOCH_B07_side_choice_locked
        clr_country_flag = KOCH_B07_side_choice_pending
        if = {
            limit = { NOT = { has_completed_focus = KOCH_B07_choose_your_side } }
            complete_national_focus = KOCH_B07_choose_your_side
        }
        KOCH_B07_clear_inherited_side_focuses = yes
        complete_national_focus = KOCH_B07_democratic_path
        mark_focus_tree_layout_dirty = yes

        KMF = {
            clr_country_flag = KOCH_B07_democratic_side
            if = {
                limit = { NOT = { has_completed_focus = KOCH_B07_choose_your_side } }
                complete_national_focus = KOCH_B07_choose_your_side
            }
            KOCH_B07_clear_inherited_side_focuses = yes
            complete_national_focus = KOCH_B07_fascist_path
            mark_focus_tree_layout_dirty = yes
        }
        KMA = {
            clr_country_flag = KOCH_B07_democratic_side
            if = {
                limit = { NOT = { has_completed_focus = KOCH_B07_choose_your_side } }
                complete_national_focus = KOCH_B07_choose_your_side
            }
            KOCH_B07_clear_inherited_side_focuses = yes
            complete_national_focus = KOCH_B07_imperial_path
            mark_focus_tree_layout_dirty = yes
        }

        # This runs after armies, leaders, wars and state ownership are set.
        if = {
            limit = { is_ai = no }
            if = {
                limit = { has_country_flag = KOCH_B07_player_chose_fascists }
                KMF = { change_tag_from = B07 }
            }
            else_if = {
                limit = { has_country_flag = KOCH_B07_player_chose_imperial }
                KMA = { change_tag_from = B07 }
            }
        }
    }
}
