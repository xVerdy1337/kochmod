# Always call in B11's country scope.
KOCH_B11_initialize = {
    if = {
        limit = { NOT = { has_country_flag = KOCH_B11_initialized } }
        set_country_flag = KOCH_B11_initialized
        set_variable = { KOCH_podik_charge = 0 }
        set_variable = { KOCH_vamos_influence = 50 }
        set_variable = { KOCH_gamos_influence = 50 }
        add_ideas = KOCH_B11_last_percent
    }
}
KOCH_B11_normalize_council = {
    if = { limit = { has_country_flag = KOCH_B11_single_chair has_country_flag = KOCH_B11_vamos_won } set_variable = { KOCH_vamos_influence = 100 } }
    if = { limit = { has_country_flag = KOCH_B11_single_chair has_country_flag = KOCH_B11_gamos_won } set_variable = { KOCH_vamos_influence = 0 } }
    clamp_variable = { var = KOCH_vamos_influence min = 0 max = 100 }
    set_variable = { KOCH_gamos_influence = 100 }
    subtract_from_variable = { KOCH_gamos_influence = var:KOCH_vamos_influence }
}
KOCH_B11_shift_vamos = {
    add_to_variable = { KOCH_vamos_influence = 10 }
    KOCH_B11_normalize_council = yes
}
KOCH_B11_shift_gamos = {
    add_to_variable = { KOCH_vamos_influence = -10 }
    KOCH_B11_normalize_council = yes
}
KOCH_B11_calculate_drain = {
    hidden_effect = {
        set_variable = { KOCH_podik_daily_drain = 0.5 }
        if = { limit = { has_completed_focus = KOCH_B11_SACRED_MANUAL } set_variable = { KOCH_podik_daily_drain = 0.3 } }
        if = { limit = { has_war = yes } add_to_variable = { KOCH_podik_daily_drain = 0.2 } }
        if = { limit = { has_country_flag = KOCH_B11_battery_saving } add_to_variable = { KOCH_podik_daily_drain = -0.2 } }
        clamp_variable = { var = KOCH_podik_daily_drain min = 0.1 max = 1 }
        set_variable = { KOCH_podik_days_to_crisis = var:KOCH_podik_charge }
        subtract_from_variable = { KOCH_podik_days_to_crisis = 20 }
        clamp_variable = { var = KOCH_podik_days_to_crisis min = 0 max = 100 }
        divide_variable = { KOCH_podik_days_to_crisis = var:KOCH_podik_daily_drain }
        if = {
            limit = { OR = { has_country_flag = KOCH_B11_charging has_country_flag = KOCH_B11_device_broken has_country_flag = KOCH_B11_device_missing } }
            set_variable = { KOCH_podik_days_to_crisis = 0 }
        }
    }
}
KOCH_B11_start_charging = {
    set_country_flag = KOCH_B11_charging
    set_variable = { KOCH_podik_charge_days = 3 }
}
KOCH_B11_daily_charge = {
    if = {
        limit = { has_country_flag = KOCH_B11_podik_path NOT = { has_country_flag = KOCH_B11_device_confiscated } }
        KOCH_B11_calculate_drain = yes
        if = {
            limit = { NOT = { has_country_flag = KOCH_B11_device_broken } NOT = { has_country_flag = KOCH_B11_device_missing } }
            if = {
                limit = { has_country_flag = KOCH_B11_auto_charge NOT = { has_country_flag = KOCH_B11_charging } check_variable = { KOCH_podik_charge < 35 } }
                KOCH_B11_start_charging = yes
            }
            if = {
                limit = { has_country_flag = KOCH_B11_charging }
                if = { limit = { has_completed_focus = KOCH_B11_SACRED_MANUAL } add_to_variable = { KOCH_podik_charge = 30 } }
                else = { add_to_variable = { KOCH_podik_charge = 20 } }
                subtract_from_variable = { KOCH_podik_charge_days = 1 }
                if = {
                    limit = { OR = { check_variable = { var = KOCH_podik_charge_days value = 0 compare = less_than_or_equals } check_variable = { var = KOCH_podik_charge value = 100 compare = greater_than_or_equals } } }
                    clr_country_flag = KOCH_B11_charging
                    set_variable = { KOCH_podik_charge_days = 0 }
                }
            }
            else = { subtract_from_variable = { KOCH_podik_charge = var:KOCH_podik_daily_drain } }
        }
        KOCH_B11_update_charge = yes
    }
}
KOCH_B11_update_charge = {
    # Previewing an effect does not simulate earlier variable assignments.
    hidden_effect = {
        clamp_variable = { var = KOCH_podik_charge min = 0 max = 100 }
        if = {
            limit = { has_country_flag = KOCH_B11_podik_path }
            if = {
                limit = { has_country_flag = KOCH_B11_device_confiscated }
                set_variable = { KOCH_podik_charge = var:KOCH_confiscated_charge }
                clr_country_flag = KOCH_B11_charging
                set_variable = { KOCH_podik_charge_days = 0 }
                remove_ideas = KOCH_B11_golden_charge
                remove_ideas = KOCH_B11_low_battery
                remove_ideas = KOCH_B11_brownout
                remove_ideas = KOCH_B11_empty_battery
            }
            else = {
            remove_ideas = KOCH_B11_low_battery
            remove_ideas = KOCH_B11_empty_battery
            if = {
                limit = { OR = { has_country_flag = KOCH_B11_device_broken has_country_flag = KOCH_B11_device_missing } }
                set_variable = { KOCH_podik_charge = 0 }
                clr_country_flag = KOCH_B11_charging
                set_variable = { KOCH_podik_charge_days = 0 }
                remove_ideas = KOCH_B11_golden_charge
                add_ideas = KOCH_B11_brownout
            }
            else = {
                if = {
                    limit = { check_variable = { var = KOCH_podik_charge value = 100 compare = greater_than_or_equals } }
                    clr_country_flag = KOCH_B11_charging
                    set_variable = { KOCH_podik_charge_days = 0 }
                }
                if = {
                    limit = { check_variable = { var = KOCH_podik_charge value = 90 compare = greater_than_or_equals } }
                    add_ideas = KOCH_B11_golden_charge
                    if = {
                        limit = { NOT = { has_country_flag = KOCH_B11_first_golden_age } }
                        set_country_flag = KOCH_B11_first_golden_age
                        country_event = { id = KOCH_PODIK.11025 days = 1 }
                    }
                }
                else = { remove_ideas = KOCH_B11_golden_charge }
                if = {
                    limit = { check_variable = { var = KOCH_podik_charge value = 40 compare = greater_than_or_equals } }
                    remove_ideas = KOCH_B11_brownout
                    clr_country_flag = KOCH_B11_charge_crisis
                }
                else_if = {
                    limit = { check_variable = { KOCH_podik_charge < 20 } }
                    if = {
                        limit = { NOT = { has_country_flag = KOCH_B11_charge_crisis } }
                        set_country_flag = KOCH_B11_charge_crisis
                        if = { limit = { check_variable = { KOCH_podik_charge > 0 } } country_event = { id = KOCH_PODIK.11007 days = 1 } }
                    }
                    if = {
                        limit = { check_variable = { KOCH_podik_charge = 0 } }
                        remove_ideas = KOCH_B11_brownout
                        add_ideas = KOCH_B11_empty_battery
                        if = {
                            limit = { NOT = { has_country_flag = KOCH_B11_empty_warning } }
                            set_country_flag = KOCH_B11_empty_warning
                            country_event = { id = KOCH_PODIK.11050 days = 1 }
                        }
                    }
                    else = { add_ideas = KOCH_B11_brownout }
                }
                else_if = {
                    limit = { has_country_flag = KOCH_B11_charge_crisis }
                    add_ideas = KOCH_B11_brownout
                }
                else = {
                    remove_ideas = KOCH_B11_brownout
                    add_ideas = KOCH_B11_low_battery
                    if = {
                        limit = { NOT = { has_country_flag = KOCH_B11_low_warning } NOT = { has_country_flag = KOCH_B11_charging } }
                        set_country_flag = KOCH_B11_low_warning
                        country_event = { id = KOCH_PODIK.11049 days = 1 }
                    }
                }
                if = { limit = { check_variable = { var = KOCH_podik_charge value = 60 compare = greater_than_or_equals } } clr_country_flag = KOCH_B11_low_warning }
                if = { limit = { check_variable = { var = KOCH_podik_charge value = 20 compare = greater_than_or_equals } } clr_country_flag = KOCH_B11_empty_warning }
            }
        }
            }

        KOCH_B11_calculate_drain = yes
    }
    custom_effect_tooltip = KOCH_B11_charge_rules_tt
}
