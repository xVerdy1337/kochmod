on_actions = {
    on_startup = {
        effect = {
            B07 = { KOCH_B07_begin_crisis = yes }
        }
    }
    on_daily_B07 = {
        effect = {
            # Initialize strength when loading a pre-war save from an older version.
            if = {
                limit = { has_country_flag = KOCH_B07_crisis_active }
                KOCH_B07_initialize_balance = yes
            }
            KOCH_B07_check_tension = yes
            if = {
                limit = {
                    has_country_flag = KOCH_B07_crisis_active
                    NOT = { has_country_flag = KOCH_B07_civil_war_started }
                    NOT = { has_country_flag = KOCH_B07_crisis_resolved }
                }
                add_to_variable = { KOCH_B07_tension_days = 1 }
                if = {
                    limit = { check_variable = { KOCH_B07_tension_days > 2 } }
                    set_variable = { KOCH_B07_tension_days = 0 }
                    add_to_variable = { KOCH_B07_tension = 1 }
                    KOCH_B07_check_tension = yes
                }
            }
        }
    }
}
